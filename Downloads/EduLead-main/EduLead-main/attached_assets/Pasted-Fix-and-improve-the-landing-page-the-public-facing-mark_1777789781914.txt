Add this reusable utility at the top of the landing page component file:

function useScrollAnimation(selector, animationOptions = {}) {
  useEffect(() => {
    const elements = document.querySelectorAll(selector);
    // Set initial state
    elements.forEach(el => {
      el.style.opacity = '0';
      el.style.transform = 'translateY(20px)';
    });

    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            anime({
              targets: entry.target,
              opacity: [0, 1],
              translateY: [20, 0],
              duration: 480,
              easing: 'easeOutQuart',
              ...animationOptions,
            });
            observer.unobserve(entry.target);
          }
        });
      },
      { threshold: 0.12 }
    );
    elements.forEach(el => observer.observe(el));
    return () => observer.disconnect();
  }, []);
}
