import anime from 'animejs/lib/anime.es.js';

/**
 * Fade + slide up a single element or selector on mount.
 * Usage: pageEnter('.page-root')
 */
export function pageEnter(target) {
  anime({
    targets: target,
    opacity: [0, 1],
    translateY: [10, 0],
    duration: 320,
    easing: 'easeOutQuad',
  });
}

/**
 * Stagger-animate a list of children into view.
 * Usage: staggerIn('.lead-card')
 */
export function staggerIn(targets, delay = 0) {
  anime({
    targets,
    opacity: [0, 1],
    translateY: [16, 0],
    delay: anime.stagger(60, { start: delay }),
    duration: 280,
    easing: 'easeOutQuad',
  });
}

/**
 * Animate a number counting up from 0 to `endValue`.
 * Writes the value into `obj.value` and calls `onUpdate(obj.value)` each frame.
 * Usage: countUp(0, 87, v => setDisplayScore(Math.round(v)))
 */
export function countUp(startValue = 0, endValue = 100, onUpdate, duration = 900) {
  const obj = { value: startValue };
  anime({
    targets: obj,
    value: endValue,
    duration,
    easing: 'easeOutCubic',
    update() {
      onUpdate(Math.round(obj.value));
    },
  });
}

/**
 * Animate a progress bar fill from 0 to `percent`.
 * Pass a ref to the fill element.
 * Usage: animateBar(fillRef.current, 72)
 */
export function animateBar(fillEl, percent, duration = 700) {
  if (!fillEl) return;
  anime({
    targets: fillEl,
    width: [`0%`, `${percent}%`],
    duration,
    easing: 'easeOutQuart',
  });
}

/**

const variantBase = {
  default:     { background: 'var(--surface-base)',   border: '1px solid var(--border-default)', boxShadow: 'var(--shadow-sm)' },
  elevated:    { background: 'var(--surface-base)',   border: '1px solid transparent',           boxShadow: 'var(--shadow-md)' },
  flat:        { background: 'var(--surface-subtle)', border: '1px solid var(--border-subtle)',   boxShadow: 'none' },
  interactive: { background: 'var(--surface-base)',   border: '1px solid var(--border-default)', boxShadow: 'var(--shadow-sm)', cursor: 'pointer' },
};

const padMap = { sm: 16, md: 24, lg: 32 };

export default function Card({ variant = 'default', padding = 'md', noPadding = false, children, onClick, style = {}, ...props }) {
  const isInteractive = variant === 'interactive' || !!onClick;

  const handleMouseEnter = (e) => {
    if (!isInteractive) return;
    anime({ targets: e.currentTarget, translateY: -2, boxShadow: '0 8px 20px rgba(0,0,0,0.08)', duration: 180, easing: 'easeOutQuad' });
  };
  const handleMouseLeave = (e) => {
    if (!isInteractive) return;
    anime({ targets: e.currentTarget, translateY: 0, boxShadow: '0 1px 3px rgba(0,0,0,0.06)', duration: 180, easing: 'easeOutQuad' });
  };

  return (
    <div
      style={{
        borderRadius: 'var(--r-lg)',
        padding: noPadding ? 0 : padMap[padding],
        transition: 'border-color var(--dur-base) var(--ease)',
        ...variantBase[variant],
        ...style,
      }}
      onMouseEnter={handleMouseEnter}
      onMouseLeave={handleMouseLeave}
      onClick={onClick}
      {...props}
    >
      {children}
    </div>
  );
}
